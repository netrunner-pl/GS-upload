Skąd wziąć adres Webhooka do Google Sheets?

Jednorazowa konfiguracja w Twoim arkuszu (zajmuje 2 minuty).

1

Otwórz Arkusz Google

W menu u góry kliknij:
Rozszerzenia → Apps Script.

2

Wklej poniższy kod

Zastąp treść w pliku Kod.gs i kliknij Zapisz (Ctrl+S).

3

Wdróż jako Web App

Niebieski przycisk Wdróż → Nowe wdrożenie, wybierz Aplikacja internetowa.

⚠️ BARDZO WAŻNY KROK W APPS SCRIPT:

W oknie wdrożenia ustaw: Kto ma dostęp (Who has access) → Każdy (Anyone). Bez tego Google zablokuje zapytania ze strony WWW!

Ulepszony kod do wklejenia w Google Apps Script (Kod.gs): 
// 1. WKLEJ TUTAJ SWOJE ID LUB CAŁY LINK DO ARKUSZA (w cudzysłowie ""):
var SPREADSHEET_ID = "TUTAJ_WKLEJ_ID_LUB_CALY_LINK";

// Funkcja pomocnicza czyszcząca ID z linku
function pobierzIdArkusza() {
  if (!SPREADSHEET_ID || SPREADSHEET_ID === "TUTAJ_WKLEJ_ID_LUB_CALY_LINK") return null;
  var match = SPREADSHEET_ID.match(/[-\w]{25,}/);
  return match ? match[0] : SPREADSHEET_ID.trim();
}

// 2. FUNKCJA DO TESTOWANIA W EDYTORZE (kliknij 'Uruchom' przy tej funkcji):
function testujPolaczenie() {
  Logger.log("--- TEST ROZPOCZĘTY ---");
  var id = pobierzIdArkusza();
  Logger.log("Wykryte ID: " + id);
  if (!id) {
    throw new Error("Wklej najpierw ID lub link w linijce: var SPREADSHEET_ID = '...';");
  }
  var ss = SpreadsheetApp.openById(id);
  Logger.log("✓ Otwarto arkusz: " + ss.getName());
  var sheet = ss.getSheetByName("Form Responses 1") || ss.getSheetByName("Odpowiedzi z formularza 1") || ss.getSheets()[0];
  Logger.log("✓ Zakładka: " + sheet.getName());
  sheet.appendRow([new Date().toLocaleString(), "TEST_MARKA", "TEST_MODEL", 999]);
  Logger.log("✓ SUKCES! Wiersz testowy pojawił się w Twoim arkuszu!");
}

// 3. ODBIORNIK DANYCH Z APLIKACJI WWW:
function doPost(e) {
  try {
    var id = pobierzIdArkusza();
    var ss = id ? SpreadsheetApp.openById(id) : SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      throw new Error("Nie znaleziono arkusza! Upewnij się, że wpisałeś SPREADSHEET_ID.");
    }
    var sheet = ss.getSheetByName("Form Responses 1") || ss.getSheetByName("Odpowiedzi z formularza 1") || ss.getSheets()[0];
    
    if (!e || !e.postData) {
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", msg: "Brak danych POST (test GET/manualny)" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = JSON.parse(e.postData.contents);
    var rows = data.rows || data.records || [];
    if (rows.length > 0) {
      for (var i = 0; i < rows.length; i++) {
        sheet.appendRow(rows[i]);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length, sheet: sheet.getName() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    console.error("BŁĄD W SKRYPCIE:", err.toString());
    throw err;
  }
}
